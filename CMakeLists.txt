cmake_minimum_required(VERSION 3.16)
project(codex-killer VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt5 5.12 REQUIRED COMPONENTS Core Widgets Sql Concurrent)
add_library(scope_core src/proc.cpp src/sessions.cpp src/terminate.cpp)
target_include_directories(scope_core PUBLIC src)
target_link_libraries(scope_core PUBLIC Qt5::Core Qt5::Sql)
target_compile_options(scope_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(codex-killer src/main.cpp src/window.cpp packaging/icons.qrc)
target_link_libraries(codex-killer PRIVATE scope_core Qt5::Widgets Qt5::Concurrent)
target_compile_options(codex-killer PRIVATE -Wall -Wextra -Wpedantic)
include(GNUInstallDirs)
install(TARGETS codex-killer RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/io.github.M1racleShih.CodexKiller.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES packaging/codex-killer.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
foreach(size 16 24 32 48 64 128 256 512)
  install(FILES packaging/icons/${size}/codex-killer.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/${size}x${size}/apps)
endforeach()
include(CTest)
if(BUILD_TESTING)
  find_package(Qt5 REQUIRED COMPONENTS Test)
  add_executable(scope_fixture tests/fixture.cpp)
  set_target_properties(scope_fixture PROPERTIES OUTPUT_NAME codex RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/fixtures")
  add_executable(scope_tests tests/test_core.cpp)
  target_link_libraries(scope_tests PRIVATE scope_core Qt5::Test)
  target_compile_definitions(scope_tests PRIVATE FIXTURE_PATH="$<TARGET_FILE:scope_fixture>")
  add_dependencies(scope_tests scope_fixture)
  add_test(NAME core COMMAND scope_tests)
  add_executable(scope_ui_tests tests/test_ui.cpp src/window.cpp packaging/icons.qrc)
  target_link_libraries(scope_ui_tests PRIVATE scope_core Qt5::Widgets Qt5::Concurrent Qt5::Test)
  add_dependencies(scope_ui_tests scope_fixture)
  target_compile_definitions(scope_ui_tests PRIVATE FIXTURE_PATH="$<TARGET_FILE:scope_fixture>")
  add_test(NAME ui COMMAND scope_ui_tests)
  set_tests_properties(ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME desktop COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_desktop.py)
  endif()
endif()
